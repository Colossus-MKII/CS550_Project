"""Extra PREFIX-only memory audit for the SAME four implementation videos.

No new H/R follow-up branches or sample selection. Rebuild the frozen prefixes
and historical treatments, then fingerprint actual neural output/memory tensors
separately from prompt coordinates. A different prompt log alone must not be
called a poisoned neural memory. Fresh output required.
"""

import argparse
import gc
from pathlib import Path

from . import configure_runtime
from .common import digest, load_predictions, write_json
from .dataset import read_manifest, load_ground_truth
from .history_check import read, native_prefix, construct_histories
from .recovery_state import (
    SAM_TARGET_KEYS,
    state_digest,
    target_history,
    synchronized_sam_offload,
)
from .runtime import checked_weight, environment


def neural_memory(history):
    """Only encoded outputs, not point coordinates or temporal bookkeeping."""
    fields = (
        "maskmem_features",
        "maskmem_pos_enc",
        "obj_ptr",
        "pred_masks",
        "object_score_logits",
    )
    return {
        kind: {
            frame: {k: value[k] for k in fields if k in value}
            for frame, value in outputs.items()
        }
        for kind, outputs in history["output_dict_per_obj"].items()
    }


def summarize_history(history, tc, t0):
    encoded = neural_memory(history)
    per_frame = {
        kind: {str(t): state_digest(value) for t, value in outputs.items()}
        for kind, outputs in encoded.items()
    }
    return {
        "container_sha256": {
            key: state_digest(history[key]) for key in SAM_TARGET_KEYS
        },
        "encoded_outputs_sha256": state_digest(encoded),
        "encoded_frame_sha256": per_frame,
        "historical_cue_encoded_sha256": state_digest(
            encoded["cond_frame_outputs"].get(tc)
        ),
        "last_pre_t0_encoded_sha256": state_digest(
            encoded["cond_frame_outputs"].get(
                t0 - 1, encoded["non_cond_frame_outputs"].get(t0 - 1)
            )
        ),
        "encoded_fields": sorted(
            {
                k
                for outputs in encoded.values()
                for entry in outputs.values()
                for k in entry
            }
        ),
    }


def execute(run, output):
    import torch
    from sam2.build_sam import build_sam2_video_predictor

    run, output = Path(run), Path(output)
    if output.exists():
        raise FileExistsError("New audit output required")
    plan = read(run / "plan.json")
    summary = read(run / "summary.json")
    if len(summary["branches"]) != 56 or summary["status"] not in {
        "running",
        "inference_completed",
    }:
        raise ValueError("First complete all frozen 56 branch computations")
    assert summary["plan_sha256"] == digest(run / "plan.json")
    if not torch.cuda.is_available() or "3080" not in torch.cuda.get_device_name(0):
        raise RuntimeError("Only local RTX3080")
    configure_runtime()
    environment("cuda:0", 550)
    weight, sha = checked_weight("sam2.1_hiera_tiny.pt")
    assert sha == summary["weight_sha256"]
    output.mkdir(parents=True)
    records = []
    predictor = build_sam2_video_predictor(
        "configs/sam2.1/sam2.1_hiera_t.yaml",
        str(weight),
        device="cuda:0",
        apply_postprocessing=True,
    )
    predictor.fill_hole_area = 0
    predictor.add_all_frames_to_correct_as_cond = True
    videos, rows, _ = read_manifest(
        plan["data_root"], [e["video_id"] for e in plan["events"]]
    )
    with torch.inference_mode(), torch.autocast(
        "cuda", dtype=torch.bfloat16
    ), synchronized_sam_offload(predictor):
        for event, video, row in zip(plan["events"], videos, rows):
            source = Path(event["baseline"]) / str(event["video_id"])
            det = load_predictions(source / "YOLO")
            gt = load_ground_truth(plan["data_root"], row)
            reference = {m: load_predictions(source / (m + "0")) for m in ("A", "B")}
            common, early, prefix, traces = native_prefix(
                video,
                det,
                reference,
                event["configuration"],
                event["t0"],
                event["history_step"],
                predictor,
            )
            histories, audits = construct_histories(
                event, common, early, prefix, traces, det, gt, output
            )
            for audit in audits:
                original = read(
                    run
                    / event["event_id"]
                    / "history"
                    / (audit["model"] + "-" + audit["history"])
                    / "audit.json"
                )
                assert (
                    original == audit
                ), "Prefix-only audit did not reproduce original history exactly"
            bid = event["anchors"]["B0"]["prediction_id"]
            values = {
                h: summarize_history(
                    target_history(histories[("B", h)][0].state, bid),
                    event["history_step"],
                    event["t0"],
                )
                for h in ("accurate", "polluted")
            }
            assert values["accurate"][
                "encoded_fields"
            ], "No actual neural tensors audited"
            changed = {
                "container_changed": {
                    key: values["accurate"]["container_sha256"][key]
                    != values["polluted"]["container_sha256"][key]
                    for key in SAM_TARGET_KEYS
                },
                "encoded_outputs_changed": values["accurate"]["encoded_outputs_sha256"]
                != values["polluted"]["encoded_outputs_sha256"],
                "last_pre_t0_encoded_outputs_changed": values["accurate"][
                    "last_pre_t0_encoded_sha256"
                ]
                != values["polluted"]["last_pre_t0_encoded_sha256"],
            }
            records.append(
                {
                    "event_id": event["event_id"],
                    "SAM": values,
                    "comparison": changed,
                    "original_history_exactly_reproduced": True,
                }
            )
            write_json(
                output / "memory-audit.json", {"status": "running", "events": records}
            )
            print("AUDIT", event["event_id"], changed, flush=True)
            del common, early, histories
            gc.collect()
            torch.cuda.empty_cache()
    write_json(
        output / "memory-audit.json",
        {
            "status": "passed",
            "events": records,
            "role": "four prefix-only numerical verification replays; NO new follow-up samples or branches",
            "model_weight_sha256": sha,
            "source_sha256": digest(Path(__file__)),
        },
    )


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--run", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    execute(args.run, args.output)
